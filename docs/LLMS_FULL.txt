## Framework Integration Code Patterns

### 1. Spring Boot Controller Integration
```kotlin
@RestController
@RequestMapping("/api/users")
class UserController {

    @PostMapping
    fun createUser(@ValidValix @RequestBody user: CreateUserRequest): ResponseEntity<UserResponse> {
        return ResponseEntity.ok(userService.create(user))
    }
}
```

### 2. Ktor Pipeline Interceptor
```kotlin
install(ValixKtorPlugin) {
    onValidationError { call, errors ->
        call.respond(HttpStatusCode.BadRequest, mapOf("errors" to errors))
    }
}
```

### 3. Jetpack Compose UI Form State
```kotlin
@Composable
fun RegistrationScreen() {
    val formState = rememberValixForm(initialValue = RegisterForm("", ""))

    ValidatedTextField(
        value = formState.value.email,
        onValueChange = { formState.update { copy(email = it) } },
        error = formState.getFieldError("email")
    )
}
```
